local S = game.ServerScriptService.Services
local H, T, W, M, V = require(S.PoliticalHierarchyService), require(S.TerritoryMapService), require(S.WarfareService), require(S.MilitaryService), require(S.VictoryService)
local a, b, c = H.GetRulerIdForUserId(-1), H.GetRulerIdForUserId(-2), H.GetRulerIdForUserId(-3)
assert(a and b and c)
task.wait(11)
-- Mechanical result cannot transfer an Unclaimed target after ownership revision.
local ok, stale = W.DeclareWar(a, "riverholt")
assert(ok, stale)
assert(M.SetCommit(a, stale, 450))
assert(M.LockCommit(a, stale))
assert(M.IsMechanicallyResolved(stale))
assert(T.SetOwner("riverholt", b))
assert(T.SetOwner("riverholt", nil))
assert(M.DebugAdvance(6))
assert(T.GetOwnerRulerId("riverholt") == nil and M.GetInspect(a).ledgerOk)
print("UC_VIC_MECHANICAL_STALE_PASS")
local fealtyOk, request = H.OfferFealty(b, a)
assert(fealtyOk, request)
assert(H.AcceptFealty(request, a))
assert(not W.DeclareWar(b, "greyfen"), "Subject authority must remain unchanged")
-- Greyfen borders subject Northvale but neither root Westmere nor Ashwood.
local borderOk, conquest = W.DeclareWar(a, "greyfen")
assert(borderOk, conquest)
assert(M.SetCommit(a, conquest, 450))
assert(M.LockCommit(a, conquest))
assert(M.DebugAdvance(6))
assert(T.GetOwnerRulerId("greyfen") == a)
print("UC_VIC_SUBJECT_BORDER_ROOT_CONQUEST_PASS")
-- Set up 11 greater-realm holdings without ending the match.
for _, land in T.GetMapDefinition().territories do
 if land.id ~= "northvale" then assert(T.SetOwner(land.id, nil)) end
end
local hold = { "westmere", "ashwood", "greyfen", "highmarch", "frostfield", "sunbarrow", "mossbank", "ironlea", "duskfen", "goldmere" }
for _, id in hold do assert(T.SetOwner(id, a)) end
assert(V.GetSnapshot().counts[a] == 11 and not V.GetSnapshot().ended)
-- Keep a second, mechanically resolved conquest in flight near the threshold.
local xOk, x = W.DeclareWar(a, "ravenreach")
assert(xOk, x)
assert(M.DebugSetTroops(a, 2000))
assert(M.SetCommit(a, x, 600))
assert(M.LockCommit(a, x))
local lastOk, last = W.DeclareWar(a, "stoneford")
assert(lastOk, last)
assert(M.SetCommit(a, last, 600))
assert(M.LockCommit(a, last))
assert(M.DebugAdvance(6))
local final = V.GetSnapshot()
assert(final.ended and final.winnerRootId == a and final.controlled == 12 and final.total == 16)
assert(#W.GetSnapshot().wars == 0 and M.GetInspect(a).committed == 0 and M.GetInspect(a).ledgerOk)
assert(T.CountOwnedByRulers({a, b}) == 12, "Second simultaneous outcome must not transfer after victory")
assert(#final.members == 2, "Subject shares winning realm")
assert(not T.SetOwner("brightfen", c))
assert(not W.DeclareWar(c, "lowmarsh"))
assert(not M.Recruit(a, 3))
assert(not M.SetCommit(a, last, 1))
assert(not M.LockCommit(a, last))
assert(not H.SetTitleRank(a, 4))
assert(not H.OfferFealty(c, a))
assert(not require(S.DiplomacyService).Offer(a, c, "Alliance"))
assert(not require(S.EconomyService).TrySpend(a, 1))
assert(V.Evaluate() == final and V.GetSnapshot().endedAt == final.endedAt)
local gold = require(S.EconomyService).GetBalance(a)
task.wait(7)
assert(require(S.EconomyService).GetBalance(a) == gold)
print("UC_VIC_FINAL_CONQUEST_FREEZE_PASS", game:GetService("HttpService"):JSONEncode(final))
