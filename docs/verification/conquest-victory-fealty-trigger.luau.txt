-- Separate fresh Local Server: joining a vassal's holdings crosses the threshold.
local S = game.ServerScriptService.Services
local H, T, V = require(S.PoliticalHierarchyService), require(S.TerritoryMapService), require(S.VictoryService)
local a, b, c = H.GetRulerIdForUserId(-1), H.GetRulerIdForUserId(-2), H.GetRulerIdForUserId(-3)
assert(a and b and c)
for _, land in T.GetMapDefinition().territories do assert(T.SetOwner(land.id, nil)) end
assert(H.SetTitleRank(a, 3)); assert(H.SetTitleRank(b, 2))
for i, land in T.GetMapDefinition().territories do
 if i <= 10 then assert(T.SetOwner(land.id, a)) elseif i <= 12 then assert(T.SetOwner(land.id, b)) end
end
assert(not V.GetSnapshot().ended)
local ok, request = H.OfferFealty(b, a)
assert(ok, request)
assert(H.AcceptFealty(request, a))
assert(V.GetSnapshot().ended and V.GetSnapshot().winnerRootId == a and V.GetSnapshot().controlled == 12)
print("UC_VIC_FEALTY_TRIGGER_PASS")
