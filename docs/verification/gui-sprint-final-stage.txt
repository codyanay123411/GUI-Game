		end)
		if ok and typeof(decoded) == "table" then

local function scaleFromCentroid(polygon: { { x: number, y: number } }, towardCenter: number)
	local cx, cy = MapGeometry.Centroid(polygon)
	local out = {}
	for _, p in polygon do
		table.insert(out, {
			x = p.x + (cx - p.x) * towardCenter,
			y = p.y + (cy - p.y) * towardCenter,
		})
	end
	return out
end

		end
		local loZ = minY - pad + halfZ
		local hiZ = maxY + pad - halfZ

		else
			lookZ = midY
		end
		camera.CFrame = CFrame.lookAt(Vector3.new(lookX, h, lookZ), Vector3.new(lookX, 0, lookZ), Vector3.new(0, 0, -1))

		local my = lookZ + (v - 0.5) * 2 * halfZ

	colorize()

		end

	intent:FireServer("Ready")
	render()

local function prettyMessage(raw: string): string
	local trimmed = string.gsub(raw, "^%s+", "")
	trimmed = string.gsub(trimmed, "%s+$", "")
	if string.sub(trimmed, 1, 1) == "{" or string.sub(trimmed, 1, 1) == "[" then
		local ok, decoded = pcall(function()
			return HttpService:JSONDecode(trimmed)
		end)
		if ok and typeof(decoded) == "table" then
			local lines = {}
			local function expand(values: any, depth: number)
				local keys = {}
				for key in values do table.insert(keys, key) end
				table.sort(keys, function(a, b) return tostring(a) < tostring(b) end)
				for _, key in keys do
					local value = values[key]
					local prefix = string.rep("  ", depth) .. tostring(key) .. ": "
					if typeof(value) == "table" and depth < 12 then
						table.insert(lines, prefix)
						expand(value, depth + 1)
					else
						table.insert(lines, prefix .. tostring(value))
					end
				end
			end
			expand(decoded, 0)
			return table.concat(lines, "\n")
		end
	end
	return raw
end

			if token == noticeToken then
				resultToast.Visible = false
			end
		end)
	end

	local function rulerById(id: string): any?
		for _, ruler in hierarchy.rulers do
			if ruler.rulerId == id then
				return ruler
			end
		end
		return nil
	end

	local function myRuler(): any?
		for _, ruler in hierarchy.rulers do
			if tonumber(ruler.userId) == player.UserId then
				return ruler
			end
		end
		return nil
	end

	local function ownerRoot(territoryId: string): string?
		local ownerId = ownership[territoryId]
		if not ownerId or ownerId == "" then
			return nil
		end
		local owner = rulerById(ownerId)
		if owner and typeof(owner.rootRulerId) == "string" and owner.rootRulerId ~= "" then
			return owner.rootRulerId
		end
		return ownerId
	end

	local function hasNap(a: string, b: string): boolean
		local keyA = if a < b then a else b
		local keyB = if a < b then b else a
		for _, row in diplomacy.agreements do
			if row.leftRulerId == keyA and row.rightRulerId == keyB and row.nap == true then
				return true
			end
		end
		return false
	end

	local function greaterRealmMemberIds(root: any): { string }
		local byId: { [string]: any } = {}
		for _, ruler in hierarchy.rulers do
			if typeof(ruler.rulerId) == "string" then
				byId[ruler.rulerId] = ruler
			end
		end
		local ids = { root.rulerId }
		local seen: { [string]: boolean } = { [root.rulerId] = true }
		local queue = { root.rulerId }
		local i = 1
		while i <= #queue do
			local current = byId[queue[i]]
			i += 1
			if current and typeof(current.vassalIds) == "table" then
				for _, vassalId in current.vassalIds do
					if typeof(vassalId) == "string" and seen[vassalId] ~= true then
						seen[vassalId] = true
						table.insert(ids, vassalId)
						table.insert(queue, vassalId)
					end
				end
			end
		end
		if typeof(root.descendantIds) == "table" then
			for _, id in root.descendantIds do
				if typeof(id) == "string" and seen[id] ~= true then
					seen[id] = true
					table.insert(ids, id)
				end
			end
		end
		return ids
	end

	local function ownedByGreaterRealm(root: any): { string }
		local members: { [string]: boolean } = {}
		for _, id in greaterRealmMemberIds(root) do
			members[id] = true
		end
		local owned = {}
		for territoryId, ownerId in ownership do
			if typeof(ownerId) == "string" and ownerId ~= "" and members[ownerId] == true then
				table.insert(owned, territoryId)
			end
		end
		return owned
	end

	local function isAdjacentForRoot(root: any, targetId: string): boolean
		local mapDef = MapCatalog.GetDefault()
		return WarfareAdjacency.IsTargetAdjacentToOwned(mapDef.territories, targetId, ownedByGreaterRealm(root))
	end

	local function territoryName(id: string): string
		local mapDef = MapCatalog.GetDefault()
		for _, t in mapDef.territories do
			if t.id == id then
				if typeof(t.displayName) == "string" and t.displayName ~= "" then
					return t.displayName
				end
				return id
			end
		end
		return id
	end

	local function rulerName(id: string): string
		local ruler = rulerById(id)
		if ruler and typeof(ruler.displayName) == "string" and ruler.displayName ~= "" then
			return ruler.displayName
		end
		return id
	end

	local function formatWarPair(war: any): string
		return string.format(
			"%s vs %s — %s",
			rulerName(war.attackerRootId),
			rulerName(war.defenderRootId),
			territoryName(war.targetTerritoryId)
		)
	end

	local function cooldownEndsAt(territoryId: string): number?
		for _, row in cooldowns do
			if row.territoryId == territoryId then
				local endsAt = tonumber(row.endsAt)
				if endsAt ~= nil and endsAt > workspace:GetServerTimeNow() then
					return endsAt
				end
				return nil
			end
		end
		return nil
	end

	local function isContested(territoryId: string): boolean
		for _, war in wars do
			if war.targetTerritoryId == territoryId then
				return true
			end
		end
		return false
	end

	local function localRootId(): string?
		return PlayerRelevantView.LocalRootRulerId(hierarchy.rulers, player.UserId)
	end

	local function relevantWars(): { any }
		return PlayerRelevantView.FilterWarsForLocalRoot(wars, localRootId())
	end

	local function warsBetween(rootA: string, rootB: string): { any }
		local matches = {}
		for _, war in wars do
			local atk = war.attackerRootId
			local def = war.defenderRootId
			if (atk == rootA and def == rootB) or (atk == rootB and def == rootA) then
				table.insert(matches, war)
			end
		end
		return matches
	end

	local refresh: () -> ()
	local cooldownUiToken = 0

	local function scheduleSelectedCooldownRefresh()
		cooldownUiToken += 1
		local token = cooldownUiToken
		local selected = selectedValue.Value
		if selected == "" then
			return
		end
		local endsAt = cooldownEndsAt(selected)
		if endsAt == nil then
			return
		end
		local remaining = endsAt - workspace:GetServerTimeNow()
		if remaining <= 0 then
			return
		end
		-- Slightly after expiry so declaration is not enabled a frame early vs server prune.
		task.delay(remaining + 0.05, function()
			if token ~= cooldownUiToken then
				return
			end
			refresh()
		end)
	end

	refresh = function()
		local me = myRuler()
		local selected = selectedValue.Value
		local bits = {}
		if not me then
			table.insert(bits, "Waiting for server state...")
		else
			if me.independent ~= true then
				table.insert(bits, "Subjects cannot independently declare external war.")
			end
			local visible = relevantWars()
			if #visible == 0 then
				table.insert(bits, "No active wars.")
			elseif #visible == 1 then
				table.insert(bits, "Active war: " .. formatWarPair(visible[1]))
			else
				local parts = {}
				for _, war in visible do
					table.insert(parts, formatWarPair(war))
				end
				table.insert(bits, "Active wars: " .. table.concat(parts, "; "))
			end
		end

		declareBtn.Text = "Declare War"
		local canDeclare = false
		local defRoot: string? = nil
		if selected == "" then
			declareBtn.Active = false
			PrototypeGuiTheme.applyPrimaryButton(declareBtn, false)
		elseif me and me.independent ~= true then
			declareBtn.Active = false
			PrototypeGuiTheme.applyPrimaryButton(declareBtn, false)
		elseif me then
			local ownerId = ownership[selected]
			if not ownerId or ownerId == "" then
				declareBtn.Active = false
				PrototypeGuiTheme.applyPrimaryButton(declareBtn, false)
			elseif ownerRoot(selected) == me.rulerId then
				declareBtn.Active = false
				PrototypeGuiTheme.applyPrimaryButton(declareBtn, false)
			else
				defRoot = ownerRoot(selected)
				if defRoot and hasNap(me.rulerId, defRoot) then
					declareBtn.Text = "NAP blocks war"
					declareBtn.Active = false
					PrototypeGuiTheme.applyPrimaryButton(declareBtn, false)
				elseif isContested(selected) then
					declareBtn.Text = "Already contested"
					declareBtn.Active = false
					PrototypeGuiTheme.applyPrimaryButton(declareBtn, false)
					table.insert(bits, "Territory is already contested.")
				elseif cooldownEndsAt(selected) then
					declareBtn.Text = "Post-war cooldown"
					declareBtn.Active = false
					PrototypeGuiTheme.applyPrimaryButton(declareBtn, false)
					table.insert(bits, "Territory is on post-war cooldown.")
				elseif not isAdjacentForRoot(me, selected) then
					declareBtn.Text = "Not adjacent"
					declareBtn.Active = false
					PrototypeGuiTheme.applyPrimaryButton(declareBtn, false)
					table.insert(bits, "Cannot declare war: target not adjacent to the greater realm.")
				else
					canDeclare = true
					local pairWars = warsBetween(me.rulerId, defRoot :: string)
					if #pairWars > 0 then
						declareBtn.Text = "Declare Additional War for " .. territoryName(selected)
					else
						declareBtn.Text = "Declare War for " .. territoryName(selected)
					end
					declareBtn.Active = true
					PrototypeGuiTheme.applyDangerButton(declareBtn)
				end
			end
		else
			declareBtn.Active = false
			PrototypeGuiTheme.applyPrimaryButton(declareBtn, false)
		end

		if selected ~= "" then
			local selName = territoryName(selected)
			if me and defRoot and canDeclare then
				local pairWars = warsBetween(me.rulerId, defRoot)
				if #pairWars > 0 then
					local existingNames = {}
					for _, war in pairWars do
						table.insert(existingNames, territoryName(war.targetTerritoryId))
					end
					table.insert(bits, "Existing war target: " .. table.concat(existingNames, ", "))
					local selectedIsExisting = false
					for _, name in existingNames do
						if name == selName then
							selectedIsExisting = true
							break
						end
					end
					if selectedIsExisting then
						table.insert(bits, "Selected: " .. selName)
					else
						table.insert(bits, "Selected new target: " .. selName)
					end
				else
					table.insert(bits, "Selected: " .. selName)
				end
			else
				table.insert(bits, "Selected: " .. selName)
			end
		end

		warStatus.Text = table.concat(bits, "\n")
		declareBtn.TextSize = 12
		scheduleSelectedCooldownRefresh()
	end

	declareBtn.MouseButton1Click:Connect(function()
		if not declareBtn.Active then
			return
		end
		local selected = selectedValue.Value
		if selected ~= "" then
			intent:FireServer("DeclareWar", selected)
		end
	end)

	selectedValue:GetPropertyChangedSignal("Value"):Connect(refresh)

	push.OnClientEvent:Connect(function(kind: any, payload: any)
		if kind == "Notice" and typeof(payload) == "string" then
			showNotice(payload)
			return
		end
		if kind ~= "State" then
			return
		end
		local data = decodeState(payload)
		if not data then
			warn("[Warfare] State decode failed")
			return
		end
		wars = data.wars or {}
		cooldowns = data.cooldowns or {}
		refresh()
	end)

	hierarchyPush.OnClientEvent:Connect(function(kind: any, payload: any)
		if kind ~= "State" then
			return
		end
		local data = decodeState(payload)
		if data and typeof(data.rulers) == "table" then
			hierarchy.rulers = data.rulers
			refresh()
		end
	end)

	territoryPush.OnClientEvent:Connect(function(kind: any, payload: any)
		if kind ~= "State" then
			return
		end
		local data = decodeState(payload)
		if not data or typeof(data.territories) ~= "table" then
			return
		end
		ownership = {}
		for _, row in data.territories do
			if typeof(row.territoryId) == "string" then
				ownership[row.territoryId] = if row.hasOwner == true then tostring(row.ownerRulerId) else ""
			end
		end
		refresh()
	end)

	diplomacyPush.OnClientEvent:Connect(function(kind: any, payload: any)
		if kind ~= "State" then
			return
		end
		local data = decodeState(payload)
		if data and typeof(data.agreements) == "table" then
			diplomacy.agreements = data.agreements
			refresh()
		end
	end)

	refresh()
	intent:FireServer("Ready")
	print("[Warfare] controller started; Ready sent")
end

return WarfareController

local function firstPlayer(value: any): Player?
	if typeof(value) == "Instance" and value:IsA("Player") then
		return value
	end
	if typeof(value) == "table" then
		local first = (value :: any)[1]
		if typeof(first) == "Instance" and first:IsA("Player") then
			return first
		end
	end
	return nil

function AdminBridge.ResolveRulerId(ref: any): (string?, string?)
	if ref == nil then
		return nil, "Need a player, rulerId, or unclaimed."
	end
	local player = firstPlayer(ref)
	if player then
		local mapped = Hierarchy.GetRulerIdForUserId(player.UserId)
		if mapped then
			return mapped, nil
		end
		return nil, "No ruler for player " .. player.Name
	end
	if typeof(ref) ~= "string" then
		ref = tostring(ref)
	end
	local s = ref :: string
	if s == "" then
		return nil, "Need a player, rulerId, or unclaimed."
	end
	local snap = Hierarchy.GetSnapshot()
	for _, ruler in snap.rulers do
		if ruler.rulerId == s then
			return s, nil
		end
	end
	local asUser = tonumber(s)
	if asUser then
		local mapped = Hierarchy.GetRulerIdForUserId(asUser)
		if mapped then
			return mapped, nil
		end
	end
	local lower = string.lower(s)
	for _, p in Players:GetPlayers() do
		if string.lower(p.Name) == lower or string.lower(p.DisplayName) == lower then
			local mapped = Hierarchy.GetRulerIdForUserId(p.UserId)
			if mapped then
				return mapped, nil
			end
			return nil, "No ruler for player " .. p.Name
		end
	end
	for _, ruler in snap.rulers do
		if typeof(ruler.displayName) == "string" and string.lower(ruler.displayName) == lower then
			return ruler.rulerId, nil
		end
	end
	return nil, "Could not resolve ruler: " .. s
end

local function inspectWars(): (boolean, string)
	local snap = Warfare.GetSnapshot()
	local wars = snap.wars or {}
	local cooldowns = snap.cooldowns or {}
	local lines = { string.format("Active wars: %d", #wars) }
	for _, war in wars do
		table.insert(
			lines,
			string.format(
				"%s\n  attacker: %s\n  defender: %s\n  target: %s",
				tostring(war.warId),
				rulerLabel(war.attackerRootId),
				rulerLabel(war.defenderRootId),
				tostring(war.targetTerritoryId)
			)
		)
	end
	if #cooldowns > 0 then
		table.insert(lines, string.format("Post-war cooldowns: %d", #cooldowns))
		for _, row in cooldowns do
			table.insert(lines, string.format("  %s until %s", tostring(row.territoryId), tostring(row.endsAt)))
		end
	end
	if #wars == 0 and #cooldowns == 0 then
		return true, "No active wars."
	end
	return true, table.concat(lines, "\n")
end

--[[
	actorUserId: Player.UserId for Kohl/live commands.
	nil = Studio/server BindableFunction (already server-only).
]]
local function pushFeedback(actorUserId: number?, operation: string, ok: boolean, message: string)
	local remotes = ReplicatedStorage:FindFirstChild("Remotes")
	local event = remotes and remotes:FindFirstChild("AdminFeedbackEvent")
	if not (event and event:IsA("RemoteEvent")) then
		return
	end
	local payload = {
		ok = ok,
		operation = operation,
		message = message,
	}
	if actorUserId ~= nil then
		local player = Players:GetPlayerByUserId(actorUserId)
		if player then
			event:FireClient(player, payload)
		end
		return
	end
	for _, player in Players:GetPlayers() do
		if AdminConfig.IsGameTester(player.UserId) then
			event:FireClient(player, payload)
		end
	end
end

function AdminBridge.Execute(actorUserId: number?, operation: string, a: any, b: any, c: any): (boolean, string)
	local function done(ok: boolean, message: string): (boolean, string)
		pushFeedback(actorUserId, if typeof(operation) == "string" then operation else "Unknown", ok, message)
		return ok, message
	end
	if actorUserId ~= nil then
		if not AdminConfig.IsGameTester(actorUserId) then
			print("[Admin] denied userId", actorUserId, operation)
			return done(fail("Not a GameTester."))
		end
	end
	if typeof(operation) ~= "string" then
		return done(fail("Need operation."))
	end
	local op = operation
	print("[Admin] Execute", actorUserId or "studio", op, a, b, c)
	if op == "Help" then
		return done(true, AdminBridge.HelpText())
	elseif op == "InspectWars" then
		return done(inspectWars())
	elseif op == "InspectTerritory" then
		return done(inspectTerritory(a))
	elseif op == "InspectRuler" then
		return done(inspectRuler(a))
	elseif op == "SetOwner" then
		return done(setOwner(a, b))
	elseif op == "SetPrestige" then
		return done(setPrestige(a, b))
	elseif op == "InspectGold" then
		return done(inspectGold(a))
	elseif op == "SetGold" then
		return done(setGold(a, b))
	elseif op == "AddGold" then
		return done(addGold(a, b))
	elseif op == "SpendGold" then
		return done(spendGold(a, b))
	elseif op == "InspectTroops" then
		return done(inspectTroops(a))
	elseif op == "SetTroops" then
		return done(setTroops(a, b))
	elseif op == "AddTroops" then
		return done(addTroops(a, b))
	elseif op == "InspectComms" then
		return done(true, HttpService:JSONEncode(Communications.DebugInspect()))
	elseif op == "InspectMilitary" then
		return done(inspectMilitary(a))
	elseif op == "InspectContest" then
		return done(inspectContest())
	elseif op == "ResolveWar" then
		return done(resolveWar(a, b))
	end
	local _ = c
	return done(fail("Unknown operation."))
end

return AdminBridge
