-- Runtime-only verification Script for a three-client Local Server.
local S = game.ServerScriptService.Services
local H = require(S.PoliticalHierarchyService)
local T = require(S.TerritoryMapService)
local W = require(S.WarfareService)
local M = require(S.MilitaryService)
local N = require(S.NeutralDefenseService)
local E = require(S.EconomyService)
local D = require(S.DiplomacyService)
local V = require(S.VictoryService)
local Calc = require(game.ReplicatedStorage.Shared.Util.Domination)
local passed = 0
local function check(ok, label)
 assert(ok, label)
 passed += 1
 print("UC-VIC PASS", passed, label)
end
local players = game.Players:GetPlayers()
table.sort(players, function(a, b) return a.UserId > b.UserId end)
assert(#players == 3, "Need exactly three test clients")
local a = H.GetRulerIdForUserId(players[1].UserId)
local b = H.GetRulerIdForUserId(players[2].UserId)
local c = H.GetRulerIdForUserId(players[3].UserId)
assert(a and b and c)
for _, land in T.GetMapDefinition().territories do assert(T.SetOwner(land.id, nil)) end
assert(T.SetOwner("westmere", a))
assert(T.SetOwner("northvale", b))
assert(T.SetOwner("brightfen", c))
assert(H.SetTitleRank(a, 3))
assert(H.SetTitleRank(b, 2))
assert(M.DebugSetSeed(42))
check(V.GetSnapshot().required == 12 and V.GetSnapshot().total == 16, "70 percent ceiling / full world denominator")
check(not W.DeclareWar(a, "missing"), "unknown target rejected")
check(not W.DeclareWar(a, "brightfen"), "nonadjacent target rejected")
local ok, id = W.DeclareWar(a, "ashwood")
check(ok and N.Get("ashwood").current == 300, "neutral conquest and configured militia")
check(not W.DeclareWar(b, "ashwood"), "same-target concurrency blocked")
check(not N.DebugSet("ashwood", 1), "admin defense cannot change contested militia")
for _, invalid in { -1, 1.5, math.huge, 0/0, 1001 } do
 check(not M.SetCommit(a, id, invalid), "malformed or over-Available commitment rejected")
end
check(not M.SetCommit(c, id, 10), "outsider cannot commit")
assert(M.SetCommit(a, id, 100))
check(M.GetInspect(a).committed == 100 and M.GetInspect(a).ledgerOk, "reservation uses normal ledger")
assert(M.LockCommit(a, id))
local battle = M.GetEngagementInspect(id).battle
check(battle.winner == "Defender", "understrength neutral attack loses")
check(N.Get("ashwood").current == battle.defenderSurvivors and battle.defenderSurvivors >= 0, "militia attrition persists and stays valid")
assert(M.DebugAdvance(6))
check(T.GetOwnerRulerId("ashwood") == nil and M.GetInspect(a).committed == 0 and M.GetInspect(a).ledgerOk, "failed battle preserves Unclaimed / releases survivors")
check(not W.DeclareWar(a, "ashwood"), "normal post-war cooldown")
local weakened = N.Get("ashwood").current
task.wait(11)
check(N.Get("ashwood").current == weakened, "persistent match survivors / no regeneration")
local success, capture = W.DeclareWar(a, "ashwood")
assert(success)
assert(M.SetCommit(a, capture, 600))
assert(M.LockCommit(a, capture))
local result = M.GetEngagementInspect(capture).battle
check(result.winner == "Attacker", "sufficient force wins neutral battle")
local income = E.GetIncomeGoldPerMinute(a)
assert(M.DebugAdvance(6))
check(T.GetOwnerRulerId("ashwood") == a and N.Get("ashwood") == nil, "successful capture through ordinary ownership")
check(E.GetIncomeGoldPerMinute(a) == income + 10, "normal future income / no neutral treasury")
check(M.GetInspect(a).committed == 0 and M.GetInspect(a).ledgerOk, "winning survivors returned exactly once")
local successStale, stale = W.DeclareWar(b, "greyfen")
assert(successStale)
assert(M.SetCommit(b, stale, 200))
assert(T.SetOwner("greyfen", c))
assert(T.SetOwner("greyfen", nil))
task.wait()
check(M.GetEngagementInspect(stale) == nil and M.GetInspect(b).committed == 0, "claim-unclaim revision invalidates pre-resolution conquest")
check(N.Get("greyfen").current == 300, "new Unclaimed lifecycle restores configured defense")
local timeoutOk, timeout = W.DeclareWar(a, "riverholt")
assert(timeoutOk)
assert(M.DebugAdvance(56))
check(M.GetEngagementInspect(timeout) == nil and T.GetOwnerRulerId("riverholt") == nil, "nonresponsive attacker cancellation")
local cancelOk, cancel = W.DeclareWar(b, "frostfield")
assert(cancelOk)
assert(M.SetCommit(b, cancel, 0))
assert(M.LockCommit(b, cancel))
check(M.GetEngagementInspect(cancel) == nil and N.Get("frostfield").current == 300, "lock-zero cancels with no militia loss")
-- Existing player war regression: both participants lock, private ledgers preserved.
local offerOk, nap = D.Offer(a, b, "NAP")
assert(offerOk)
assert(D.Accept(nap, b))
check(not W.DeclareWar(a, "northvale"), "existing NAP still blocks player war")
assert(D.Break(a, b, "NAP"))
local playerOk, playerWar = W.DeclareWar(a, "northvale")
assert(playerOk)
assert(M.SetCommit(a, playerWar, 20))
assert(M.SetCommit(b, playerWar, 200))
assert(M.LockCommit(a, playerWar))
assert(M.LockCommit(b, playerWar))
assert(M.DebugAdvance(6))
check(T.GetOwnerRulerId("northvale") == b and M.GetInspect(a).ledgerOk and M.GetInspect(b).ledgerOk, "player-war defense regression")
-- Pure different-map, deep-subject, independence and denominator checks.
local map = { victory = { type = "Domination", territoryControlPercent = 0.70 }, territories = {} }
local owners = {}
for i = 1, 10 do table.insert(map.territories, { id = tostring(i) }); table.insert(owners, { territoryId = tostring(i), ownerRulerId = if i <= 6 then "leaf" else "" }) end
local rulers = { { rulerId = "root", rootRulerId = "root", independent = true }, { rulerId = "leaf", rootRulerId = "root", independent = false } }
check(Calc.Evaluate(map, rulers, owners).winner == nil, "6 of 10 does not win despite all claimed land belonging to realm")
owners[7].ownerRulerId = "leaf"
check(Calc.Evaluate(map, rulers, owners).winner.rootId == "root", "different map ceiling / descendant ownership")
rulers[2].rootRulerId = "leaf"; rulers[2].independent = true
check(Calc.Evaluate(map, rulers, owners).counts.root == nil, "independence removes former-root holdings")
check(not pcall(Calc.Evaluate, { victory = { type = "Domination", territoryControlPercent = 0 }, territories = map.territories }, rulers, owners), "invalid threshold rejected")
print("UC_VIC_STAGE_ONE_PASS", passed, a, b, c)
