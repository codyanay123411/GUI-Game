function WarfareService.DeclareWar(attackerRootId: string, targetTerritoryId: string): (boolean, string)
	if MatchState.IsEnded() then return false, "Match ended." end
	if typeof(targetTerritoryId) ~= "string" or targetTerritoryId == "" then
		return false, "Cannot declare war: invalid target."
	end
	local attacker = findRuler(attackerRootId)
	if not isIndependentRoot(attacker) then
		return false, "Subjects cannot independently declare external war."
	end
	local mapDef = Territory.GetMapDefinition()
	local exists = false
	for _, t in mapDef.territories do
		if t.id == targetTerritoryId and t.conquerable ~= false then
			exists = true
			break
		end
	end
	if not exists then
		return false, "Cannot declare war: unknown territory."
	end
	local ownerId = Territory.GetOwnerRulerId(targetTerritoryId)
	local neutral = if not ownerId then Neutral.Get(targetTerritoryId) else nil
	if isOwnedByRealm(targetTerritoryId, attackerRootId) then
		return false, "Cannot declare war: target belongs to your realm."
	end
	local defenderRoot = ownerRootId(targetTerritoryId) or ""
	local defender = findRuler(defenderRoot)
	if not neutral and not isIndependentRoot(defender) then
		return false, "Cannot declare war: defender is not an independent realm."
	end
	if not isGreaterRealmAdjacent(attackerRootId, targetTerritoryId) then
		return false, "Cannot declare war: target not adjacent."
	end
	if not neutral and Diplomacy.HasNonAggressionPact(attackerRootId, defenderRoot) then
		return false, "Cannot declare war: Non-Aggression Pact active."
	end
	if activeWarOnTerritory(targetTerritoryId) then
		return false, "Territory is already contested."
	end
	if isTerritoryOnCooldown(targetTerritoryId) then
		return false, "Territory is on post-war cooldown."
	end
	for _, war in warsById do
		if war.status == "Active" and war.attackerRootId == attackerRootId and war.targetTerritoryId == targetTerritoryId then
			return false, "Cannot declare war: a matching war is already active."
		end
	end
	local warId = HttpService:GenerateGUID(false)
	local created: War = {
		warId = warId,
		attackerRootId = attackerRootId,
		defenderRootId = defenderRoot,
		targetTerritoryId = targetTerritoryId,
		status = "Active",
		targetKind = if neutral then "Unclaimed" else "Player",
		targetRevision = Territory.GetOwnershipRevision(targetTerritoryId),
		neutralDefenders = if neutral then neutral.current else nil,
	}
	warsById[warId] = created
	militaryModule().OnWarCreated(created)
	pushState()
	noticeRulerId(attackerRootId, if neutral then "Conquest begun. Commit Troops in Military." else "War declared.")
	noticeRulerId(defenderRoot, "War declared against your realm.")
	return true, warId
end

function WarfareService.CancelWar(warId: string, reason: string)
	local war = warsById[warId]
	if not war or war.status ~= "Active" then
		return
	end
	local attacker = war.attackerRootId
	local defender = war.defenderRootId
	endWar(war, reason)
	pushState()
	noticeRulerId(attacker, "War cancelled (" .. reason .. ").")
	noticeRulerId(defender, "War cancelled (" .. reason .. ").")
end

function WarfareService.ApplyMilitaryOutcome(warId: string, outcome: string): (boolean, string)
	if MatchState.IsEnded() then return false, "Match ended." end
	local war = warsById[warId]
	if not war or war.status ~= "Active" then
		return false, "War is no longer active."
	end
	if outcome ~= "Attacker" and outcome ~= "Defender" then
		return false, "Unknown resolution."
	end
	if war.targetKind == "Unclaimed" and (not targetStillOriginalDefender(war) or not isIndependentRoot(findRuler(war.attackerRootId))) then
		endWar(war, "stale neutral conquest")
		pushState()
		return false, "Conquest stale; no transfer."
	end
	if outcome == "Defender" then
		local attacker = war.attackerRootId
		local defender = war.defenderRootId
		endWar(war, "defender victory")
		pushState()
		noticeRulerId(attacker, "War ended: defender victory. Ownership unchanged.")
		noticeRulerId(defender, "War ended: defender victory. Ownership unchanged.")
		return true, "Defender victory."
	end
	local attackerId = war.attackerRootId
	local defenderId = war.defenderRootId
	local targetId = war.targetTerritoryId
	if not targetStillOriginalDefender(war) then
		endWar(war, "attacker victory stale target")
		pushState()
		noticeRulerId(attackerId, "War ended: attacker victory. Target no longer belongs to the original defender.")
		noticeRulerId(defenderId, "War ended: attacker victory. Target no longer belongs to the original defender.")
		return true, "Attacker victory recorded; territory was not transferred."
	end
	endWar(war, "attacker victory")
	local ok, err = Territory.SetOwner(targetId, attackerId)
	if not ok then
		pushState()
		return false, err
	end
	pushState()
	noticeRulerId(attackerId, "War ended: attacker victory. You now hold " .. targetId .. ".")
	noticeRulerId(defenderId, "War ended: attacker victory. " .. targetId .. " was lost.")
	return true, "Attacker victory."
end

function WarfareService.IsTargetCurrent(warId: string): boolean
	local war = warsById[warId]
	return war ~= nil and targetStillOriginalDefender(war)
end

local function cancelInvolving(rulerId: string, reason: string)
	local removed = false
	local toEnd = {}
	for _, war in warsById do
		if war.status == "Active" and (war.attackerRootId == rulerId or war.defenderRootId == rulerId) then
			table.insert(toEnd, war)
		end
	end
	local Military = militaryModule()
	for _, war in toEnd do
		if Military.IsMechanicallyResolved(war.warId) then
			-- Mechanical result stays authoritative through presentation.
		else
			Military.UnwindPreResolution(war.warId)
			local other = if war.attackerRootId == rulerId then war.defenderRootId else war.attackerRootId
			endWar(war, reason)
			noticeRulerId(other, "War cancelled (" .. reason .. ").")
			removed = true
		end
	end
	if removed then
		noticeRulerId(rulerId, "External wars ended (" .. reason .. ").")
		pushState()
	end
end

local function invalidateStaleWars()
	local removed = false
	local Military = militaryModule()
	for _, war in warsById do
		if war.status == "Active" and not targetStillOriginalDefender(war) then
			if Military.IsMechanicallyResolved(war.warId) then
				-- Do not reopen a mechanically final battle.
			else
				Military.UnwindPreResolution(war.warId)
				endWar(war, "stale after ownership change")
				noticeRulerId(war.attackerRootId, "War ended: target no longer belongs to the original defender.")
				noticeRulerId(war.defenderRootId, "War ended: target no longer belongs to the original defender.")
				removed = true
			end
		end
	end
	if removed then
		pushState()
	end
end

local function resolveBattle(eng: Engagement)
	if eng.mechanicallyResolved then
		return
	end
	local atk = eng.attacker.lockedAmount
	local def = eng.defender.lockedAmount
	if typeof(atk) ~= "number" or typeof(def) ~= "number" then
		return
	end
	if atk <= 0 then
		diag("attacker locked 0; cancel", eng.warId)
		cancelEngagement(eng, "attacker committed no force")
		noticeRulerId(eng.attackerRootId, "Attack cancelled: you committed 0 Troops.")
		return
	end
	if eng.targetKind == "Unclaimed" and (not Warfare.IsTargetCurrent(eng.warId) or not (findRuler(eng.attackerRootId) and findRuler(eng.attackerRootId).independent)) then
		cancelEngagement(eng, "stale target")
		return
	end
	local battle = MilitaryCombat.Resolve(atk, def, makeRng())
	eng.battle = battle
	eng.mechanicallyResolved = true
	eng.presentationUntil = now() + MilitaryConfig.PresentationSeconds
	applyCasualties(eng, battle)
	if eng.targetKind == "Unclaimed" then Neutral.ApplySurvivors(eng.targetTerritoryId, eng.targetRevision, battle.defenderSurvivors) end
	diag("battle resolved", eng.warId, battle.winner, battle.band or "unopposed")
	pushAllPlayers()
end

local function tryLock(eng: Engagement, side: SideState, amount: number)
	if side.locked then
		return
	end
	side.locked = true
	side.lockedAmount = amount
	side.intended = amount
	side.interacted = true
	if side == eng.attacker and amount <= 0 then
		diag("attacker locked 0; cancel", eng.warId)
		cancelEngagement(eng, "attacker committed no force")
		noticeRulerId(eng.attackerRootId, "Attack cancelled: you committed 0 Troops.")
		return
	end
	if eng.attacker.locked and eng.defender.locked then
		resolveBattle(eng)
	else
		pushAllPlayers()
	end
end

function MilitaryService.Recruit(rulerId: string, troopAmount: any): (boolean, string)
	if MatchState.IsEnded() then return false, "Match ended." end
	if totalByRuler[rulerId] == nil then
		return false, "No troop state."
	end
	if typeof(troopAmount) ~= "number" or troopAmount ~= troopAmount then
		return false, "Recruit amount must be a whole number."
	end
	if troopAmount ~= math.floor(troopAmount) then
		return false, "Recruit amount must be a whole number."
	end
	if troopAmount <= 0 then
		return false, "Recruit amount must be greater than 0."
	end
	if troopAmount < MilitaryConfig.TroopsPerGold then
		return false, "Minimum recruit is 3 Troops."
	end
	if troopAmount % MilitaryConfig.TroopsPerGold ~= 0 then
		return false, "Recruit amount must be a multiple of 3 Troops."
	end
	local goldCost = troopAmount / MilitaryConfig.TroopsPerGold
	local ok, err = Economy.TrySpend(rulerId, goldCost)
	if not ok then
		return false, err
	end
	totalByRuler[rulerId] += troopAmount
	pushAllPlayers()
	return true, string.format("Recruited %d Troops for %d Gold.", troopAmount, goldCost)
end

function MilitaryService.SetCommit(rulerId: string, warId: string, amountAny: any): (boolean, string)
	if MatchState.IsEnded() then return false, "Match ended." end

		return false, "Commitment must be a whole number."

		return false, "Commitment cannot be negative."

	local available = availableOf(rulerId) + current
	if amountAny > available then

	end

		return true, string.format("Commitment set to %d.", amountAny)

local function tickEngagements()
	if MatchState.IsEnded() then return end
	local t = now()
	local snapshot = {}
	for _, eng in engagements do
		table.insert(snapshot, eng)
	end
	for _, eng in snapshot do
		if MatchState.IsEnded() then break end
		if not eng.concluded then
			if eng.mechanicallyResolved then
				if eng.presentationUntil and t >= eng.presentationUntil then
					concludeEngagement(eng)
				end
			else
				if t >= eng.startedAt + MilitaryConfig.CommitWindowSeconds then
					if not eng.attacker.locked and eng.attacker.interacted and typeof(eng.attacker.intended) == "number" then
						tryLock(eng, eng.attacker, eng.attacker.intended)
					end
					if stillOpen(eng) and not eng.defender.locked and eng.defender.interacted and typeof(eng.defender.intended) == "number" then
						tryLock(eng, eng.defender, eng.defender.intended)
					end
				end
				if stillOpen(eng) and t >= eng.startedAt + MilitaryConfig.HardTimeoutSeconds then
					if not eng.attacker.locked then
						cancelEngagement(eng, "attacker did not commit")
					elseif not eng.defender.locked then
						tryLock(eng, eng.defender, 0)
					end
				end
			end
		end
	end
end

	local queue = { root.rulerId }
	local seen: { [string]: boolean } = { [root.rulerId] = true }
	local i = 1

		i += 1
		if current then

local function validateFealty(fromRulerId: string, toRulerId: string): (boolean, string)
	if fromRulerId == toRulerId then
		return false, "Cannot offer vassalage to yourself."
	end
	local fromRuler = rulersById[fromRulerId]
	local toRuler = rulersById[toRulerId]
	if not fromRuler or not toRuler then
		return false, "Ruler no longer exists."
	end
	if fromRuler.liegeId == toRulerId then
		return false, "That direct relationship already exists."
	end
	if not RankRules.CanBecomeVassalOf(fromRuler.titleRank, toRuler.titleRank) then
		return false, "Cannot become the vassal of a lower-ranked ruler."
	end
	if isInSubtree(fromRulerId, toRulerId) then
		return false, "That change would create a political cycle."
	end
	return true, ""
end

function PoliticalHierarchyService.OfferFealty(fromRulerId: string, toRulerId: string): (boolean, string)
	if MatchState.IsEnded() then return false, "Match ended." end
	local ok, err = validateFealty(fromRulerId, toRulerId)
	if not ok then
		return false, err
	end
	supersedePair(fromRulerId, toRulerId)
	local requestId = HttpService:GenerateGUID(false)
	local created = now()
	local lifetime = Config.VassalageRequestLifetimeSeconds
	if typeof(lifetime) ~= "number" or lifetime <= 0 then
		lifetime = 30
	end
	local request: FealtyRequest = {
		requestId = requestId,
		fromRulerId = fromRulerId,
		toRulerId = toRulerId,
		createdAt = created,
		expiresAt = created + lifetime,
		expireThread = nil,
	}
	request.expireThread = task.delay(lifetime, function()
		local expired = requestsById[requestId]
		if expired then
			expired.expireThread = nil
			local fromRuler = rulersById[expired.fromRulerId]
			removeRequest(requestId)
			pushState()
			noticeRuler(fromRuler, "Request expired.")
		end
	end)
	requestsById[requestId] = request
	pushState()
	return true, requestId
end

function PoliticalHierarchyService.DeclineFealty(requestId: string, actorRulerId: string?): (boolean, string)
	if MatchState.IsEnded() then return false, "Match ended." end
	local request = requestsById[requestId]
	if not request then
		return false, "Request is no longer valid."
	end
	if now() >= request.expiresAt then
		removeRequest(requestId)
		pushState()
		return false, "Request has expired."
	end
	if actorRulerId and actorRulerId ~= request.toRulerId then
		return false, "Only the prospective liege may decline this request."
	end
	local fromRuler = rulersById[request.fromRulerId]
	local toRuler = rulersById[request.toRulerId]
	removeRequest(requestId)
	pushState()
	if toRuler then
		noticeRuler(toRuler, "Vassalage declined.")
	end
	if fromRuler and toRuler then
		noticeRuler(fromRuler, string.format("%s declined your vassalage offer.", toRuler.displayName))
	end
	return true, "Vassalage declined."
end

function PoliticalHierarchyService.AcceptFealty(requestId: string, actorRulerId: string?): (boolean, string)
	if MatchState.IsEnded() then return false, "Match ended." end
	local request = requestsById[requestId]
	if not request then
		return false, "Request is no longer valid."
	end
	if now() >= request.expiresAt then
		removeRequest(requestId)
		pushState()
		return false, "Request has expired."
	end
	if actorRulerId and actorRulerId ~= request.toRulerId then
		return false, "Only the prospective liege may accept this request."
	end
	local ok, err = validateFealty(request.fromRulerId, request.toRulerId)
	if not ok then
		removeRequest(requestId)
		pushState()
		return false, err

		return

		return

	notice(player, "Unknown action.")

		if not war or typeof(war.warId) ~= "string" then
			return false
		end
		if war.phase ~= "commit" or war.locked == true then
			return false
		end
		local parsed = parseWholeNonNegative(commitBox.Text)
		if parsed == nil then
			local text = commitBox.Text
			if text == "" then
				fieldStatus = ""
			else
				fieldStatus = "Invalid amount (not a whole number >= 0)"
			end
			return false
		end
		if parsed > maxCommitFor(war) then
			fieldStatus = "Invalid: over Available"
			if lastSentByWar[war.warId] == parsed and not force then
				return false
			end
			lastSentByWar[war.warId] = parsed
			intent:FireServer("SetCommit", war.warId, parsed)
			return false
		end
		if lastSentByWar[war.warId] == parsed and not force then
			fieldStatus = ""
			return true
		end
		fieldStatus = ""
		lastSentByWar[war.warId] = parsed
		intent:FireServer("SetCommit", war.warId, parsed)
		return true
	end

	local function scheduleCommitSend()
		debounceToken += 1
		local token = debounceToken
		task.delay(COMMIT_DEBOUNCE_SEC, function()
			if token == debounceToken then

			end
		end)
	end

local function scaleFromCentroid(polygon: { { x: number, y: number } }, towardCenter: number)
	local cx, cy = MapGeometry.Centroid(polygon)

local function prettyMessage(raw: string): string
	local trimmed = string.gsub(raw, "^%s+", "")
	trimmed = string.gsub(trimmed, "%s+$", "")
	if string.sub(trimmed, 1, 1) == "{" or string.sub(trimmed, 1, 1) == "[" then
		local ok, decoded = pcall(function()
			return HttpService:JSONDecode(trimmed)
		end)

			local lines = {}
			local function expand(values: any, depth: number)
				local keys = {}
				for key in values do table.insert(keys, key) end
				table.sort(keys, function(a, b) return tostring(a) < tostring(b) end)
				for _, key in keys do
					local value = values[key]
					local prefix = string.rep("  ", depth) .. tostring(key) .. ": "
					if typeof(value) == "table" and depth < 12 then
						table.insert(lines, prefix)
						expand(value, depth + 1)
					else
						table.insert(lines, prefix .. tostring(value))
					end
				end
			end
			expand(decoded, 0)
			return table.concat(lines, "\n")
		end
	end
